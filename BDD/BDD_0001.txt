CREATE TABLE monsters (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100),
    element VARCHAR(20),
    base_hp INTEGER,
    base_atk INTEGER,
    base_def INTEGER,
    base_spd INTEGER
);

CREATE TABLE runes (
    id SERIAL PRIMARY KEY,
    slot INTEGER,
    set_name VARCHAR(50),
    hp INTEGER,
    atk INTEGER,
    defense INTEGER,
    speed INTEGER,
    crit_rate INTEGER,
    crit_dmg INTEGER
);
